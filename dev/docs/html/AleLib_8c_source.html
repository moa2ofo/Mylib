24void MyLib_ProcessRecord(const MyLib_record_t *rec_pc, uint8_t multiplier_u8) {
25 uint32_t l_acc_u32;
26 uint8_t l_i_u8;
27
28 /* Check if input pointer is NULL */
29 if(rec_pc == NULL) {
30 return;
31 }
32
33 /* Switch-case structure for bounded accumulation based on multiplier_u8 */
34 switch(multiplier_u8) {
35 case 0U:
36 /* No accumulation */
37 l_acc_u32 = 0U;
38 break;
39
40 case 1U:
41 /* Single direct assignment */
42 l_acc_u32 = rec_pc->value_u32;
43 break;
44
45 default:
46 /* Bounded loop accumulation */
47 l_acc_u32 = 0U;
48 for(l_i_u8 = 0U; l_i_u8 < multiplier_u8; l_i_u8++) {
49 l_acc_u32 += rec_pc->value_u32;
50 }
51 break;
52 }

54 /* Increment global counter by accumulated value */
55 g_counter_u32 += l_acc_u32;

57 /* Invoke library function with accumulated value and multiplier address */
58 (void)MyLib_ComputeAdjustedValue_u32(l_acc_u32, (const uint16_t *)&multiplier_u8);

139uint8_t MyLib_UpdateCounter_u8(uint32_t add_u32) {
140 /* Function-local static cycle counter, persistent across calls */
141 static uint32_t l_CycleCnt_u32 = 0U;
142
143 uint32_t l_new_u32;
144 uint8_t l_ret_u8;
145
146 /* Step 1: Increment the function-static cycle counter */
147 l_CycleCnt_u32++;
148
149 /* Step 2: Check module readiness */
150 if(g_systemReady_b == false) {
151 /* System not ready, return error code 1 and perform no update */
152 return 1U;
153 }
154
155 /* Step 3: Compute tentative new counter value */
156 l_new_u32 = g_counter_u32 + add_u32;
157

171 SaturationEn_b = !SaturationEn_b;
172 }

206 /* Function-local static variables to preserve state machine context across calls */
207 static uint8_t l_State_u8 = 0U;
208 static MyLib_record_t l_LastRecord;
209 static uint32_t l_LastAdjusted_u32 = 0U;
210
211 /* Local variables */

213 uint32_t l_UpdateValue_u32 = 0U;
214
215 /* State machine execution */
216 switch(l_State_u8) {
217 case 0U: /* Init state */
218 if(rec_pc == NULL) {
219 l_ret_u8 = 1U;
220 } else {
221 MyLib_UpdateGlobalRecord(&l_LastRecord, rec_pc);
222 l_State_u8 = 1U;
223 l_ret_u8 = 0U;
224 }
225 break;
226

228 MyLib_ProcessRecord(&l_LastRecord, MYLIB_MULT_VALUE_U8);
229 l_LastAdjusted_u32 = MyLib_ComputeAdjustedValue_u32(l_LastRecord.value_u32, delta_pc_u16);

236 l_ret_u8 = MyLib_UpdateCounter_u8(l_UpdateValue_u32);
237 l_State_u8 = 0U;
238 break;

242 l_ret_u8 = 0U;
